    def load_data_csv(self):
        """Method that loads data from csv file and returns dataframe."""
        try:
            df = pd.read_csv(self.get_path(), on_bad_lines="warn")
            df.columns = df.columns.str.strip()
            return df
        except FileNotFoundError:
            raise PathError()
